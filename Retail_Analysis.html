SQL Queries Used¶

Total Sales¶

SELECT SUM(Sales) FROM RetailSales;

Sales by Category¶

SELECT Category, SUM(Sales) FROM RetailSales GROUP BY Category;

Sales by Region¶

SELECT Region, SUM(Sales) FROM RetailSales GROUP BY Region;

Top 10 Products¶

SELECT Product_Name, SUM(Sales) AS TotalSales FROM RetailSales GROUP BY Product_Name ORDER BY TotalSales DESC LIMIT 10;
